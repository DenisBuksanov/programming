# Отчет по расчетно-графической работе
## Тема: «Простейший переводчик»

---

## 1. Цель работы и техническое задание
Разработать консольное приложение на языке Си, выполняющее автоматический контекстный перевод слов в текстовом файле по заданному пользователем словарю. 

### Основные инженерные требования, реализованные в работе:
1. **Динамическое выделение и расширение памяти:** Словарь не ограничен жесткими рамками компиляции и масштабируется в процессе чтения файла.
2. **Многозначность лексем:** Возможность хранения до 5 альтернативных вариантов перевода для одного оригинального слова.
3. **Сохранение исходной разметки:** Точное удержание исходного форматирования текста (все пробелы, знаки препинания, переводы строк `\n` и табуляции остаются на своих исходных местах).
4. **Контекстный регистр:** Автоматическое распознавание и сохранение заглавных букв при переводе как для латиницы (ASCII), так и для кириллицы (двухбайтовый UTF-8).

---

## 2. Разработанная структура данных

Для эффективного связывания оригинального слова со спектром его значений в оперативной памяти была спроектирована комбинированная структура `DictRec`:

```c
typedef struct {
    char og[256];          // Оригинальное слово (ключ поиска в нижнем регистре)
    char transl[5][256];   // Статический двумерный массив альтернативных вариантов перевода
    int t_count;           // Фактическое количество загруженных вариантов перевода
} DictRec;
```

Данный подход позволяет зафиксировать максимальную длину слова в пределах 255 символов (плюс терминальный ноль \0) и оптимизировать поиск по первичному ключу og.

## 3. Подробный анализ проделанной работы и ключевых функций
В ходе выполнения расчетно-графической работы были решены четыре фундаментальные алгоритмические задачи.

1. Модуль побайтового анализа кодировки UTF-8 (make_low)
Проблема: Стандартная библиотечная функция tolower() работает исключительно в рамках однобайтовой кодировки ASCII. В современных Unix-подобных системах (включая macOS и Linux) кириллица кодируется в многобайтовом формате UTF-8, где каждая заглавная русская буква (А–Я) физически занимает в памяти ровно 2 байта.

Решение: Был разработан собственный побайтовый декодер регистров.

```c
int make_low(const char* src, char* dst) { 
    strcpy(dst, src);
    unsigned char b1 = dst[0], b2 = dst[1];
    
    // Английский алфавит в ASCII
    if (b1 >= 'A' && b1 <= 'Z') { 
        dst[0] += 32; 
        return 1; 
    }
    // Русская заглавная кириллица в UTF-8 (А-Я)
    // Символы кодируются парой байт, где первый байт равен 0xD0, а второй лежит в диапазоне 0x90-0xAF
    if (b1 == 0xD0 && b2 >= 0x90 && b2 <= 0xAF) {
        if (b2 <= 0x9F) dst[1] = b2 + 0x20; // Корректировка для диапазона букв А-П
        else { dst[0] = 0xD1; dst[1] = b2 - 0x20; } // Корректировка для диапазона букв Р-Я
        return 1; 
    }
    return 0; 
}
```

Результат работы модуля: Функция не просто копирует слово в нижнем регистре в буфер поиска dst, но и возвращает логический флаг (1 или 0), сигнализирующий вызывающей функции о том, требовалось ли трансформировать исходный регистр.

2. Модуль вывода текста в верхний регистр (fprint_upper)
Данная функция решает задачу динамического поднятия регистра первой буквы переведенного слова непосредственно перед записью в файл результатов, если исходное слово в тексте начиналось со строчной (большой) буквы.

```C
void fprint_upper(FILE *out, const char *str) {
    unsigned char b1 = str[0], b2 = str[1];
    
    // Английская строчная лексема
    if (b1 >= 'a' && b1 <= 'z') {
        fprintf(out, "%c%s", b1 - 32, str + 1);
    }
    // Русская строчная кириллица (диапазон а-п, второй байт 0xB0-0xBF)
    else if (b1 == 0xD0 && b2 >= 0xB0 && b2 <= 0xBF) {
        fprintf(out, "%c%c%s", 0xD0, b2 - 0x20, str + 2);
    } 
    // Русская строчная кириллица (диапазон р-я, второй байт 0x80-0x8F)
    else if (b1 == 0xD1 && b2 >= 0x80 && b2 <= 0x8F) {
        fprintf(out, "%c%c%s", 0xD0, b2 + 0x20, str + 2);
    } 
    else {
        fprintf(out, "%s", str); // Символ уже в верхнем регистре или не является буквой
    }
}
```

Результат работы модуля: Реализовано побайтовое восстановление заглавного регистра для букв а-я за счет обратной математической корректировки смещения байтов в кодовой таблице UTF-8.

3. Функция динамической загрузки и парсинга словаря (load_dict)
Этот модуль реализует отказоустойчивую работу с оперативной памятью компьютера («кучей» / heap) и токенизацию строк словаря.

```C
DictRec* load_dict(const char *filename, int *dict_size) {
    FILE *f = fopen(filename, "r");
    if (!f) {
        printf("Ошибка: файл словаря не найден\n");
        return NULL;
    }

    int capacity = 10, count = 0;
    DictRec *dict = malloc(capacity * sizeof(DictRec)); // Первичное выделение памяти

    if (!dict) {
        printf("Ошибка: Не удалось выделить память под словарь.\n");
        fclose(f);
        return NULL;
    }

    char line[1024];
    while (fgets(line, sizeof(line), f)) {
        char *word = strtok(line, " \n\r\t"); // Выделение ключа-оригинала
        if (!word) continue;

        int idx = -1;
        for (int i = 0; i < count; i++) {
            if (strcmp(dict[i].og, word) == 0) { idx = i; break; }
        }

        // Если ключ уникален — создаем новую запись
        if (idx == -1) {
            if (count >= capacity) { // Механизм динамического расширения
                capacity *= 2; 
                DictRec *temp = realloc(dict, capacity * sizeof(DictRec));
                if (!temp) { // Защита от утечки памяти при отказе realloc
                    printf("Ошибка: Нехватка памяти при расширении словаря. Загружено слов: %d.\n", count);
                    free(dict); fclose(f); return NULL;
                }
                dict = temp;
            }
            idx = count++;
            strcpy(dict[idx].og, word);
            dict[idx].t_count = 0;
        }

        char *trans; // Заполнение массива альтернативных вариантов перевода
        while ((trans = strtok(NULL, " \n\r\t")) && dict[idx].t_count < 5) {
            strcpy(dict[idx].transl[dict[idx].t_count++], trans);
        }
    }
    fclose(f);

    if (count == 0) {
        printf("Предупреждение: Файл словаря '%s' пуст или поврежден.\n", filename);
        free(dict); return NULL;
    }
    *dict_size = count;
    return dict;
}
```

Результат работы модуля: Решена проблема дублирования ключей (если одно слово записано в словаре на нескольких строках, переводы склеиваются в одну структуру). Внедрена концепция динамического перевыделения памяти (capacity *= 2), защищенная от утечек (memory leaks) при помощи временного буферного указателя temp.

4. Функция перевода и агрегации многозначных лексем (transl_word)
Модуль управляет логикой сопоставления слова из исходного текста с базой данных словаря.

```C
void transl_word(const char *word, DictRec *dict, int dict_size, FILE *out) {
    char search[256];
    int is_upper = make_low(word, search); // Приведение к нижнему регистру для поиска по совпадению

    for (int i = 0; i < dict_size; i++) {
        if (strcmp(search, dict[i].og) == 0) { // Линейный поиск ключа
            for (int j = 0; j < dict[i].t_count; j++) {
                if (is_upper) {
                    fprint_upper(out, dict[i].transl[j]); // Печать с сохранением заглавного регистра
                } else {
                    fprintf(out, "%s", dict[i].transl[j]); // Печать в стандартном нижнем регистре
                }
                
                if (j < dict[i].t_count - 1) {
                    fprintf(out, "/"); // Установка разделителя строго МЕЖДУ вариантами перевода
                }
            }
            return; // Успешный выход из функции
        }
    }
    fprintf(out, "%s", word); // Если перевод отсутствует, токен пишется в неизменном виде
}
```

Результат работы модуля: Обеспечен вывод многозначных понятий в формате перечисления через косую черту без появления лишних «хвостовых» слэшей.

5. Посимвольный анализатор текста (Функция main)
Основной цикл программы реализует низкоуровневый конечный автомат для разбора текста. Вместо ненадежного чтения строк (fgets) или разбиения пробелами (fscanf), которые уничтожают авторскую пунктуацию, применено посимвольное чтение через fgetc().

```C
int main(int argc, char *argv[]) {
    if (argc != 4) {
        printf("Используйте: %s <текст.txt> <словарь.txt> <результат.txt>\n", argv[0]);
        return 1;
    }

    int dict_size = 0;
    DictRec *dict = load_dict(argv[2], &dict_size);
    if (!dict) return 1;

    FILE *in = fopen(argv[1], "r");
    if (!in) {
        printf("Ошибка: Не удалось открыть файл с исходным текстом '%s'.\n", argv[1]);
        free(dict); return 1;
    }
    FILE *out = fopen(argv[3], "w");
    if (!out) {
        printf("Ошибка: Не удалось создать или открыть на запись файл результата '%s'.\n", argv[3]);
        fclose(in); free(dict); return 1;
    }

    char token[256];
    int pos = 0;
    int ch;

    // Главный цикл потокового посимвольного сканирования
    while ((ch = fgetc(in)) != EOF) {
        // Если считанный байт — английская буква (isalpha) ИЛИ часть многобайтового UTF-8 символа (ch >= 128)
        if (isalpha((unsigned char)ch) || (unsigned char)ch >= 128) {
            if (pos < 255) token[pos++] = ch; // Аккумулируем символ в текущее слово
        } else {
            // Если встречен разделитель (пробел, знак препинания, перенос строки)
            if (pos > 0) {
                token[pos] = '\0'; // Закрываем буфер валидным нуль-терминатором C-строки
                transl_word(token, dict, dict_size, out); // Отправляем изолированное слово на перевод
                pos = 0; // Сбрасываем позицию для накопления следующего токена
            }
            fputc(ch, out); // Выводим сам знак препинания, пробел или символ '\n' без изменений
        }
    }
    // Обработка пограничного условия: файл завершился словом без знака препинания на конце
    if (pos > 0) {
        token[pos] = '\0';
        transl_word(token, dict, dict_size, out);
    }

    // Каскадное закрытие дескрипторов и очистка кучи (ОЗУ)
    free(dict);
    fclose(in); 
    fclose(out);
    printf("Успех: Перевод успешно выполнен. Результат сохранен в '%s'.\n", argv[3]);
    return 0;
}
```

Результат работы модуля: Достигнуто 100% сохранение структуры исходного форматирования текста. Знаки препинания (запятые, точки, многоточия), пробелы и отступы табуляции не обрабатываются словарем, а напрямую копируются в выходной поток.
